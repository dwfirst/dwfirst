def get_memory_files():

    today = datetime.now().strftime('%Y-%m-%d')

    return os.path.join(CHRONICLES_DIR, f"full_log_{today}.txt"), os.path.join(CHRONICLES_DIR, f"index_{today}.txt")



def get_associative_context(limit=20):

    _, index_file = get_memory_files()

    if not os.path.exists(index_file): return "История сегодня пуста."

    with open(index_file, "r", encoding="utf-8") as f:

        return "".join(f.readlines()[-limit:])



def recall_full_log(target_id, date_str=None):

    if not date_str: date_str = datetime.now().strftime('%Y-%m-%d')

    log_file = os.path.join(CHRONICLES_DIR, f"full_log_{date_str}.txt")

    if not os.path.exists(log_file): return ""

    content, found = [], False

    with open(log_file, "r", encoding="utf-8") as f:

        for line in f:

            if f"[ID:{target_id}]" in line: found = True

            if found:

                content.append(line)

                if "--------------------" in line: break

    return "".join(content)



def smart_memory_search(query):

    _, index_file = get_memory_files()

    if not os.path.exists(index_file): return ""

    keywords = [w.lower() for w in query.split() if len(w) > 4]

    relevant_ids = []

    with open(index_file, "r", encoding="utf-8") as f:

        for line in f:

            if any(k in line.lower() for k in keywords):

                try:

                    res_id = line.split("ID:")[1].split(" |")[0].strip()

                    relevant_ids.append(res_id)

                except: continue

    deep_memories = ""

    for rid in relevant_ids[-2:]:

        deep_memories += f"\n[ПОДНЯТО ИЗ ГЛУБИНЫ]:\n{recall_full_log(rid)}"

    return deep_memories

    relevant_ids = []

        for line in f:

                    relevant_ids.append(res_id)
                except: continue
    deep_memories = ""

    return deep_memories
