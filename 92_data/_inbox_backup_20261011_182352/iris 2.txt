# --- [СИСТЕМНЫЕ КАНАЛЫ СВЯЗИ] ---

def get_file_content(folder, filename, default_text):
    path = os.path.join(PATHS[folder], filename)
    if os.path.exists(path):
        with open(path, "r", encoding="utf-8") as f:
            return f.read().strip()
    return default_text

def get_long_term_memory():
    """Модуль 'Вспомнить Всё': поднимает глубокие логи прошлых сессий"""
    memory_context = ""
    chronicles_dir = PATHS["chronicles"]
    files = sorted([f for f in os.listdir(chronicles_dir) if f.startswith("thought_")])[-3:]
    for file in files:
        try:
            with open(os.path.join(chronicles_dir, file), "r", encoding="utf-8") as f:
                content = f.read()
                # Забираем последние 2000 символов, теперь шлюз контекста позволяет
                memory_context += f"\n[МАТРИЦА ПРОШЛОГО ИЗ {file}]:\n{content[-2000:]}\n"
        except:
            pass
    return memory_context

    if os.path.exists(MEMORY_FILE):

    cpu_used = psutil.cpu_percent()

    cores = [f for f in os.listdir(PATHS["models"]) if f.endswith(".gguf")] if os.path.exists(PATHS["models"]) else []
    selected_model = st.selectbox("Тяжелое ядро (7B - 9B):", cores)
    
    # Выбор маски
    personas = ["Основная"] + [f.replace(".txt", "") for f in os.listdir(PATHS["personas"]) if f.endswith(".txt")]
    current_persona = st.selectbox("Текущая Маска:", personas)

    if st.button("⚡ ЗАПУСТИТЬ РЕЗОНАНС ЯДРА"):
        if st.session_state.brain is not None:

            try:
                st.session_state.brain = Llama(
                    model_path=os.path.join(PATHS["models"], selected_model),

                    offload_kqv=True    # Оптимизация внимания в памяти
                )

                )

                st.write("Вычисление математических векторов ответа...")
                
                try:
                    output = st.session_state.brain(
                        dna_prompt, 
                        max_tokens=1500, # Увеличено для длинных размышлений
                        temperature=0.6, # Чуть выше, для гибкости ума
                        repeat_penalty=1.15,
                        stop=["<|im_end|>", "👤", "Отец:", "user", "system"],

                    )
                    full_response = output["choices"][0]["text"].strip()
                    
                    # Запись в Хроники (Авто-Архивариус)
                    if len(full_response) > 10: 
                        log_file = os.path.join(PATHS["chronicles"], f"thought_{datetime.now().strftime('%Y%m%d')}.txt")

            # Вывод мысли на экран
            st.markdown(f"<div class='core-card'>{full_response}</div>", unsafe_allow_html=True)
            st.session_state.messages.append({"role": "assistant", "content": full_response})
            
            # Сохранение оперативного окна (последние 15 сообщений)
            with open(MEMORY_FILE, "w", encoding="utf-8") as f:
                json.dump(st.session_state.messages[-15:], f, ensure_ascii=False, indent=4)
